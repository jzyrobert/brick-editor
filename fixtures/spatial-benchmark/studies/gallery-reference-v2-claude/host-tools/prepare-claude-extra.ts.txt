// Lays out accepted frozen Claude v2 submissions as one-shot folders for
// `npm run gallery:publish`, with the benchmark record in result.json.
import { readFile, writeFile, mkdir, readdir } from "node:fs/promises";
import { join, resolve } from "node:path";
import { existsSync } from "node:fs";
import { createHash } from "node:crypto";
import assert from "node:assert/strict";

const root = resolve(".local/claude-v2"),
  out = resolve(".local/claude-v2-extra-publish");
const hash = (b: Uint8Array | string) =>
  createHash("sha256").update(b).digest("hex");
const read = async (p: string) => JSON.parse(await readFile(p, "utf8"));
const ledger: unknown[] = [];
const expected: Record<string, string> = {};
for (const [model, runName, transport] of [["haiku", "haiku-uncapped", "claude-native-cli/3, maxOutputTokens 128000"], ["opus", "opus-continue", "claude-native-cli/3, continued with FINAL_NOTICE"]]) {
  const m = await read(join(root, runName, "manifest.json"));
  // Cases a container restart interrupted ran again in <model>-resume.
  const results: any[] = [];
  for (const d of [runName])
    if (existsSync(join(root, d, "results.json")))
      for (const r of (await read(join(root, d, "results.json"))).results)
        results.push({ ...r, dir: join(root, d) });

  for (const r of results) {
    const dir = r.dir;
    const task = m.tasks.find((t: any) => r.caseId.startsWith(t.id + "--"));
    const run = join(dir, r.caseId);
    for (const [file, sha] of Object.entries(r.files))
      assert.equal(hash(await readFile(join(run, file))), sha, file);
    // Cost and output tokens from the native turn results.
    const native = join(dir, r.caseId + "--transport", "native");
    let costUsd = 0,
      output = 0;
    for (const f of (await readdir(native)).filter((f) =>
      f.endsWith(".jsonl"),
    ))
      for (const line of (await readFile(join(native, f), "utf8")).split(
        "\n",
      )) {
        if (!line.includes('"type":"result"')) continue;
        const e = JSON.parse(line);
        costUsd += e.total_cost_usd ?? 0;
        output += e.modelUsage?.[m.model.id]?.outputTokens ?? 0;
      }
    ledger.push({
      model: m.model,
      caseId: r.caseId,
      retry: true,
      status: r.status,
      failure: r.failure ?? null,
      used: r.used,
      seconds: r.seconds,
      costUsd: Math.round(costUsd * 100) / 100,
      outputTokens: output,
      files: r.files,
    });
    if (r.status !== "submitted") continue;
    const c = await read(join(run, "compilation.json"));
    assert(c.report.ok && c.script);
    const name = `gallery-reference-v2-claude-${runName}-${task.id}`;
    const target = join(out, task.id, name);
    await mkdir(join(target, m.model.effort), { recursive: true });
    await writeFile(
      join(target, "prompt.md"),
      `Build request: ${task.brief}\n`,
    );
    await writeFile(
      join(target, m.model.effort, "build.json"),
      JSON.stringify(c.script, null, 2) + "\n",
    );
    await writeFile(
      join(target, m.model.effort, "result.json"),
      JSON.stringify(
        {
          model: m.model.id,
          runner: "claude",
          effort: m.model.effort,
          accepted: true,
          targetParts: task.targetParts,
          parts: c.report.stats.parts,
          seconds: r.seconds,
          tokens: { output, costUsd: Math.round(costUsd * 100) / 100 },
          benchmark: {
            caseId: r.caseId,
            transport,
            observationPolicy: "blind-mechanical",
            factsVersion: "spatial-part-facts/2",
            sizePolicy: "advisory",
            used: r.used,
            referenceCoverage: r.referenceCoverage,
            frozenFiles: r.files,
          },
        },
        null,
        2,
      ) + "\n",
    );
    expected[r.files["build.mpd"].slice(0, 12)] = `${model} ${task.id}`;
  }
}
await writeFile(join(out, "outcomes.json"), JSON.stringify(ledger, null, 2) + "\n");
await writeFile(
  join(out, "expected-ids.json"),
  JSON.stringify(expected, null, 2) + "\n",
);
console.log(JSON.stringify({ publishable: Object.keys(expected).length, ledger: ledger.length }));
