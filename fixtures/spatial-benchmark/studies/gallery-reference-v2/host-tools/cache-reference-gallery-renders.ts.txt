import {readFile,writeFile,mkdir,stat} from 'node:fs/promises';
import {join,resolve} from 'node:path';
import {createHash} from 'node:crypto';
import assert from 'node:assert/strict';
import {prepare,renderAll} from '../scripts/gallery-publish';
const root=resolve('.local/gallery-reference-v2-runs'),out=resolve('.local/gallery-reference-v2-render-cache');
await mkdir(out,{recursive:true});
const read=async(p:string)=>JSON.parse(await readFile(p,'utf8'));
const hash=(b:Uint8Array)=>createHash('sha256').update(b).digest('hex');
for(;;){
 const state=await read(join(root,'status.json'));const pending=[];
 for(const [model,cp] of Object.entries(state.checkpoints) as [string,string][]){
  const manifest=await read(join(cp,'manifest.json'));const results=await read(join(cp,'results.json'));
  for(const r of results.results){
   if(r.status!=='submitted')continue;
   const dir=join(out,model+'-'+r.caseId);
   if(await stat(join(dir,'complete.json')).then(()=>true,()=>false))continue;
   for(const [name,sha] of Object.entries(r.files))assert.equal(hash(await readFile(join(cp,r.caseId,name))),sha);
   const c=await read(join(cp,r.caseId,'compilation.json'));const task=manifest.tasks.find((t:any)=>r.caseId.startsWith(t.id+'--'));
   await mkdir(dir,{recursive:true});await writeFile(join(dir,'build.json'),JSON.stringify(c.script,null,2)+'\n');
   const b=await prepare({label:model+' '+r.caseId,scriptPath:join(dir,'build.json'),brief:task.brief,targetParts:task.targetParts,promptId:'cache',runner:'codex',model:manifest.model.id,effort:manifest.model.effort,agentId:'cache',agentName:'cache',generation:'cache'});
   assert.equal(b.mpd.sha,r.files['build.mpd']);pending.push({b,dir});
  }
 }
 if(pending.length){
  const images=await renderAll(pending.map(x=>x.b));
  for(const {b,dir} of pending){const hashes:Record<string,string>={};for(const [key,bytes] of Object.entries(images.get(b.id)!)){await writeFile(join(dir,key+'.webp'),bytes);hashes[key]=hash(bytes);}
   await writeFile(join(dir,'complete.json'),JSON.stringify({id:b.id,mpdSha256:b.mpd.sha,images:hashes}));
   console.log(JSON.stringify({cached:b.id,dir}));
  }
 }
 if(state.state==='complete')break;
 await new Promise(r=>setTimeout(r,30000));
}
