import {chromium,expect} from '@playwright/test';
import {readFile,writeFile} from 'node:fs/promises';
const manifest=JSON.parse(await readFile('.local/gallery-reference-v2-bundle/manifest.json','utf8'));
const browser=await chromium.launch({headless:true,args:['--no-sandbox','--use-angle=swiftshader','--enable-unsafe-swiftshader']});
try{
 const page=await browser.newPage();page.setDefaultTimeout(60000);
 const results=[];
 for(const viewport of [{width:1440,height:1000},{width:360,height:600},{width:1080,height:1800},{width:411,height:685},{width:390,height:844},{width:686,height:411}]){
  await page.setViewportSize(viewport);await page.goto('https://bricks.robertj.in/?galleryIndex=1');
  await expect(page.getByRole('heading',{name:'Build gallery',exact:true})).toBeVisible();
  const generation=page.getByRole('combobox',{name:'Prompt generation',exact:true});
  if(!await generation.isVisible())await page.getByRole('button',{name:/^Filters/}).click();
  await generation.selectOption(manifest.generation.id);
  await expect(page.locator('.gallery-response')).toHaveCount(manifest.publishable);
  const overflow=await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth+1);
  if(overflow)throw Error('Gallery has horizontal page overflow');
  await expect.poll(()=>page.locator('.gallery-stage > img').first().evaluate(i=>(i as HTMLImageElement).complete&&(i as HTMLImageElement).naturalWidth>0)).toBe(true);
  const rendered=await page.locator('.gallery-stage > img').evaluateAll(imgs=>imgs.filter(i=>(i as HTMLImageElement).complete&&(i as HTMLImageElement).naturalWidth>0).length);
  await page.screenshot({path:`.local/gallery-reference-v2-live-${viewport.width}.png`,fullPage:false});
  results.push({viewport,cards:manifest.publishable,loadedVisibleImages:rendered,horizontalOverflow:false});
 }
 await writeFile('.local/gallery-reference-v2-live-browser.json',JSON.stringify({checkedAt:new Date().toISOString(),results},null,2)+'\n');
 console.log(JSON.stringify(results));
}finally{await browser.close();}
